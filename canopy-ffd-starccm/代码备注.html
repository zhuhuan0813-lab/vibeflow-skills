1．最先看这三个位置

  1. 看操作流程：_program/start.py。从 menu 找到数字对应的函数，重点看 generate、calculate 和 report。

3．需要改什么，应该去哪里

想调整的内容应修改的位置与注意事项
本机软件路径优先使用菜单 5；路径保存在 settings.json。
舱盖变形模式与幅度FFD 的 configs/canopy_10.json；targetMaxMm 是最大表面节点位移，单位 mm。简易入口使用固定的 10 方案，改源码包后要开新批次并重新检查。
工况、网格与收敛判据CFD 的 0-caseDict/caseDict；再生成 solver.properties。简易入口锁定 50 m/s、0°、+X 和默认统计节奏，其他工况须按详细说明修改与验证。
FFD 算法、控制点或曲率约束_generate.py；当前 105 个格点、最多 27 个非零控制点。平滑边缘权重不等于已经施加曲率约束。
STAR 对象、边界或求解模型三个 Java 宏，需匹配 STAR 版本。不要只改名字而不核对对象与边界映射。
对比图与统计窗口D-post/postprocess.py，同时核对 caseDict 与宏的采样方式。默认末 150 次迭代、每 10 次采样，共 15 个样本。
重要：settings.json、canopy_10.json 和 caseDict 是 JSON，不能直接插入 # 或 // 注释。说明放在本页或源码里；配置的字段名、单位和结构保留原格式。

4．读注释时记住这些约定

  • Python / PowerShell：# 后是注释；Java：// 后是注释。CMD 的 REM 注释保持英文 ASCII，以兼容 Windows 入口，中文解释在本页。
  • 单位：FFD 原始数据为 mm，prepare_from_ffd 乘一次 0.001 变成 m；STAR 几何和网格尺寸使用 m。
  • 方向：来流与阻力为 +X，升力为 +Z；报告中的力作用于完整机身壁面。
  • 完成与收敛：最终 .sim、日志标记与 convergence.json 一起检查；仅有文件不能证明收敛，更不能证明优化有效。
  • 修改后开新批次：注释也会改变文件哈希。已有批次受输入一致性保护，请保留原结果，用新运行包重新生成。

class WorkflowError(Exception):
    pass

def read_json(path):
    # utf-8-sig 同时兼容普通 UTF-8 和 Windows 工具写入的 UTF-8 BOM。
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))

def save_json(path, value):
    # 先写本次独有的临时文件，再替换目标，减少记录写到一半的风险。
    # 这里保存的是设置和状态，不是删除或覆盖原始模型、求解结果。
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + '.tmp.' + str(os.getpid()) + '.' + uuid.uuid4().hex)
    tmp.write_text(json.dumps(value, ensure_ascii=False, indent=2), encoding='utf-8')
    tmp.replace(path)

def settings(home):
    # home 由双击入口的位置确定，用户只需保存软件路径，不用填写项目根目录。
    cfg = read_json(home / 'settings.json')
    for key in ['windows_python', 'wsl_python', 'wsl_distro', 'star']:
        if not isinstance(cfg.get(key), str) or not cfg[key].strip():
            raise WorkflowError('软件位置还没有填完整，请先选择菜单 5。')
    return cfg

def capture(args, timeout=90):
    # 用于版本/路径等轻量检查；返回标准输出，失败时带回报错供中文提示展示。
    try:
        p = subprocess.run(list(map(str, args)), capture_output=True, timeout=timeout)
    except (OSError, subprocess.TimeoutExpired) as e:
        raise WorkflowError('程序无法启动或检查超时：' + str(args[0]) + '\n' + str(e)) from e
    out = p.stdout.decode('utf-8-sig', errors='replace')
    err = p.stderr.decode('utf-8-sig', errors='replace')
    if p.returncode:
        raise WorkflowError(f'程序返回错误 {p.returncode}：{args[0]}\n' + (err or out)[-1600:])
    return out

def wsl_args(cfg, args):
    # --exec 后直接传程序和独立参数，不把用户路径拼成 bash 命令字符串。
    return ['wsl.exe', '-d', cfg['wsl_distro'], '--exec', *map(str, args)]

def wsl_path(cfg, path):
    # 同一份文件在 Windows 与 WSL 中写法不同，统一交给 wslpath 转换。
    # 取以 / 开头的路径行，避免把 WSL 启动提示误当成目录。
    output = capture(wsl_args(cfg, ['wslpath', '-a', '-u', str(Path(path).resolve())]))
    candidates = [s.strip() for s in output.splitlines() if s.strip().startswith('/')]
    if not candidates:
        raise WorkflowError('无法把项目路径转换为 WSL 路径。请检查菜单 5 的 WSL 发行版。')
    return candidates[-1]

def child(args, logfile, timeout=3600):
    # 每次执行都使用带时间戳的新日志；重试时保留上次日志，便于定位失败阶段。
    # 子程序非零退出立即中止当前流程，避免把部分生成的文件当作完成结果。
    logfile = Path(logfile)
    logfile.parent.mkdir(parents=True, exist_ok=True)
    logfile = logfile.with_name(logfile.stem + '_' + dt.datetime.now().strftime('%Y%m%d_%H%M%S_%f') + logfile.suffix)
    try:
        with logfile.open('xb') as stream:
            p = subprocess.run(list(map(str, args)), stdout=stream, stderr=subprocess.STDOUT, timeout=timeout)
    except (OSError, subprocess.TimeoutExpired) as e:
        raise WorkflowError('本步骤没有完成。已保留日志：' + str(logfile) + '\n' + str(e)) from e
    if p.returncode:
        raise WorkflowError('本步骤失败，暂不进入下一步。日志：\n' + str(logfile))

def check_windows(cfg):
    # 文件存在不代表解释器可用，因此实际读取 Python 版本和 sys.executable。
    py = Path(cfg['windows_python'])
    if not py.is_file():
        raise WorkflowError('Windows Python 路径无效，请在菜单 5 选择实际 python.exe。')
    value = json.loads(capture([py, '-c', 'import json,sys;print(json.dumps({"version":list(sys.version_info[:3]),"path":sys.executable}))']))
    if value['version'] < [3, 10, 0]:
        raise WorkflowError('需要 Python 3.10 或更新版本。')
    return 'Windows Python ' + '.'.join(map(str, value['version']))

def check_geometry(cfg):
    # 依赖在 WSL 的 Python 中导入；Windows 安装的包不会自动供 WSL 使用。
    output = capture(wsl_args(cfg, [cfg['wsl_python'], '-c', 'import numpy,scipy,pyvista,vtk,matplotlib;print("WSL geometry environment OK")']))
    if 'WSL geometry environment OK' not in output:
        raise WorkflowError('WSL 的 Python 检查没有返回预期结果。')
    return '所需的 5 个 Python 库均可导入'

def check_star(cfg):
    # -version 只验证入口与版本；许可、Java API、建网格是否成功要看实际基线。
    if not Path(cfg['star']).is_file():
        raise WorkflowError('STAR 启动文件不存在，请在菜单 5 选择 starccm+.bat。')
    # .bat needs cmd; list2cmdline follows the original driver's Windows launch convention.
    try:
        p = subprocess.run(subprocess.list2cmdline([cfg['star'], '-version']), shell=True, capture_output=True, timeout=90)
    except (OSError, subprocess.TimeoutExpired) as e:
        raise WorkflowError('STAR 版本检查没有完成：' + str(e)) from e
    output = (p.stdout + p.stderr).decode('utf-8', errors='replace')
    if p.returncode or EXPECTED_RELEASE not in output:
        raise WorkflowError('STAR 版本检查未通过。当前宏需要 2510 / 20.06.007-R8，请核对启动路径和版本。')
    return 'STAR-CCM+ 20.06.007 系列；许可与实际求解将在基线中验证'

def check_environment(home):

    cfg = settings(home)

def unpack_core(payload, root):
    # 每次生成都解压到全新批次，保留旧批次；xb 模式不允许覆盖已有文件。
    # CFD 模板放到批次/cfd，其余脚本和输入放到批次/程序，便于区分运行产物。
    root = Path(root).resolve()
    if root.exists():
        raise WorkflowError('本批次目录已存在，程序不会覆盖。')
    root.mkdir(parents=True)
    with zipfile.ZipFile(payload) as z:
        for entry in z.infolist():
            parts = Path(entry.filename).parts
            if not parts or parts[0] != 'project' or '..' in parts:
                raise WorkflowError('运行包目录不正确。')
            rel = Path(*parts[1:])
            if not rel.parts:
                continue
            if rel.parts[0] == 'cfd':
                target = root / rel
            else:
                target = root / '程序' / rel
            target = target.resolve()
            # ZIP 成员只能写入本批次目录，不能通过 ../ 等路径写到其他位置。
            if not target.is_relative_to(root):
                raise WorkflowError('运行包包含越界路径。')
            if entry.is_dir():
                target.mkdir(parents=True, exist_ok=True)
            else:
                target.parent.mkdir(parents=True, exist_ok=True)
                with target.open('xb') as out:
                    out.write(z.read(entry))

def contract(cfd):
    # 简易入口锁定已确认的工况和完整 10 方案，防止改一个数字却沿用不适配的宏。
    # 这里只做当前入口的适用范围检查，不代表 CFD 的物理精度已经验证。
    cfg = read_json(cfd / '0-caseDict/caseDict')
    if cfg['flow']['speed'] != 50 or cfg['flow']['aoa'] != 0 or cfg['flow']['direction'] != [1, 0, 0] or cfg['flow']['rotors']:
        raise WorkflowError('简易入口用于 50 m/s、0°、+X、无旋翼工况。修改工况请使用详细教程并重新验证。')
    if [x['id'] for x in cfg['variants']] != [f'{i:02d}' for i in range(1, 11)]:
        raise WorkflowError('简易入口要求完整的 10 个方案。')
    if cfg['solver']['sampleEvery'] != 10 or cfg['solver']['convergenceWindow'] != 150:
        raise WorkflowError('简易入口使用原有的 10 次采样与 150 次窗口设置。')
    return cfg

def fingerprint(cfd):
    # 为几何、配置、宏和后处理记录 SHA-256；计算前复查，防止混用修改后的输入。
    # 注释也会改变文件哈希，因此修改已有批次脚本后应建立新批次。
    paths = [cfd / '0-caseDict/caseDict', cfd / '0-caseDict/solver.properties', *sorted((cfd / 'A-geo').glob('*.stl')), *sorted((cfd / 'UDF').glob('*.java')), cfd / 'UDF/run_fresh_batch.py', cfd / 'D-post/postprocess.py']
    return {str(p.relative_to(cfd)).replace('\\', '/'): hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}

def generate(home, no_open=False):
    # 菜单 2：只做 FFD、几何验收、mm→m 转换和 STAR 命令预览，不求解流场。
    cfg = settings(home)
    print('\n正在生成原始机身和 10 组舱盖方案，本步骤不启动 CFD。')
    check_geometry(cfg)
    stamp = dt.datetime.now().strftime('批次_%Y%m%d_%H%M%S_%f')
    # 微秒时间戳让连续两次“生成模型”形成独立目录，旧结果不被覆盖。
    root = home / '结果' / stamp
    unpack_core(home / '_program/core.zip', root)
    cfd = root / 'cfd'
    contract(cfd)
    source_wsl = wsl_path(cfg, root / '程序')
    root_wsl = wsl_path(cfg, root)
    print('1/3 生成形状并检查几何……')
    # --units mm 是几何导出单位；内部 NPZ 中的坐标和位移同样保持毫米。
    child(wsl_args(cfg, [cfg['wsl_python'], source_wsl + '/ffd/run_ffd.py', '--config', source_wsl + '/ffd/configs/canopy_10.json', '--output', root_wsl + '/ffd_result', '--units', 'mm']), root / '日志/生成模型.log')
    ffd = read_json(root / 'ffd_result/run_manifest.json')
    # 只有状态完整、检查过交叉且包含十个方案，才继续生成 STAR 输入。
    if ffd['status'] != 'completed' or not ffd['intersectionsChecked'] or len(ffd['variants']) != 10:
        raise WorkflowError('FFD 检查状态不完整，暂不准备 STAR 输入。')
    print('2/3 自动转换为米，准备 STAR 输入……')
    # prepare_from_ffd.py 内部只转换一次 mm→m；这里不要手工再缩放 STL。
    child(wsl_args(cfg, [cfg['wsl_python'], source_wsl + '/prepare_from_ffd.py', '--ffd-run', root_wsl + '/ffd_result', '--cfd-root', root_wsl + '/cfd']), root / '日志/准备STAR输入.log')
    print('3/3 检查将要执行的批处理命令……')
    # --dry-run 打印宏路径与命令，不会启动 STAR 求解，也不能替代许可/API 验证。
    child([sys.executable, cfd / 'UDF/run_fresh_batch.py', '--root', cfd, '--star', cfg['star'], '--dry-run'], root / '日志/命令预览.log')
    state = {'run': stamp, 'projectHome': str(home), 'phase': '模型与STAR输入已准备_未计算CFD', 'inputHashes': fingerprint(cfd)}
    save_json(root / '批次状态.json', state)
    # 所有准备步骤完成后才更新“最近批次”，失败目录仍保留供排查。
    save_json(home / '结果/最近批次.json', {'run': stamp})
    print('\n模型生成完成，三维预览即将打开。看完后，选择 3 才会开始气动计算。')
    open_file(root / 'ffd_result/viewer.html', no_open)
    return root

def latest(home):
    # 读取最近成功准备的批次，并限制目录位于本项目的“结果”文件夹内。
    p = home / '结果/最近批次.json'
    if not p.exists():
        raise WorkflowError('还没有生成模型。请先选择菜单 2。')
    name = read_json(p).get('run', '')
    if not re.fullmatch(r'批次_\d{8}_\d{6}_\d{6}', name):
        raise WorkflowError('最近批次记录无效。请重新生成模型；已有结果会保留。')
    root = (home / '结果' / name).resolve()
    if not root.is_relative_to((home / '结果').resolve()) or not root.is_dir():
        raise WorkflowError('最近批次目录不存在。请重新生成模型。')

def validate_inputs(home, root):

def case_finished(cfd, name, require_converged=True):
    # 最终 .sim 存在还不够：同时要求历史、收敛 JSON、正常完成日志。
    # 默认必须 converged=true；诊断报告可显式允许已运行但未收敛的变形方案。
    folder = cfd / 'C-run' / name
    required = ['final_result.sim', 'convergence.json', 'history.csv', 'batch.log']
    if not all((folder / x).is_file() for x in required):
        return False
    if 'VF_STAGE=CASE_COMPLETE' not in (folder / 'batch.log').read_text(encoding='utf-8', errors='replace'):
        return False

@contextmanager
def batch_lock(root):
    # 系统文件锁阻止两个窗口同时计算同一批次；进程结束后锁自动释放。
    # lock 文件保留并不表示仍在计算，是否占用由操作系统锁状态决定。
    path = root / '计算占用.lock'
    with path.open('a+b') as stream:
        if path.stat().st_size == 0:
            stream.write(b'0')
            stream.flush()
        stream.seek(0)
        try:
            if os.name == 'nt':
                import msvcrt
                msvcrt.locking(stream.fileno(), msvcrt.LK_NBLCK, 1)
            else:
                import fcntl
                fcntl.flock(stream.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        except OSError as e:
            raise WorkflowError('这个批次正在另一个窗口计算，请回到原窗口查看。') from e
        try:
            yield
        finally:
            stream.seek(0)
            if os.name == 'nt':
                msvcrt.locking(stream.fileno(), msvcrt.LK_UNLCK, 1)
            else:
                fcntl.flock(stream.fileno(), fcntl.LOCK_UN)

  if x['mode'] not in ['raise','lower','widen','narrow','front','rear','combined']:raise ValueError('Unsupported mode: '+x['mode'])

// 第三阶段：接续初算，按窗口检查收敛，保存最终结果和可机读的判断依据。
// 达到最大迭代数也会保存结果，但 converged=false 不能解释为优化有效。

   // 窗口样本仅累计本续算阶段的数据，不包含前一宏的初算点。
   int sample=Integer.parseInt(p.getProperty("sampleEvery")), window=Integer.parseInt(p.getProperty("convergenceWindow"))/sample;
   List<Double> ds=new ArrayList<>();boolean converged=false;double drift=Double.NaN,range=Double.NaN,mass=Double.NaN,resmax=Double.NaN;
   // 以追加方式写入同一 history.csv，保留初算历史；每次前进后才记录一行。

     // 力单位 N、流量单位 kg/s；质量不平衡无量纲，避免单看残差判断收敛。
     double d=drag.getReportMonitorValue(),l=lift.getReportMonitorValue(),sf=side.getReportMonitorValue(),i=mi.getReportMonitorValue(),o=mo.getReportMonitorValue(),y=yp.getReportMonitorValue();mass=Math.abs(i+o)/Math.max(Math.abs(i),1e-12);ds.add(d);out.printf(Locale.US,"%d,%.12g,%.12g,%.12g,%.12g,%.12g,%.12g,%.12g,%.12g,%.12g%n",it,d,l,sf,drag.getPressureValue().x,drag.getShearValue().x,i,o,mass,y);out.flush();if(!Double.isFinite(d)||Math.abs(d)>1e7)throw new RuntimeException("Force divergence");

     // drift=|前窗均值-后窗均值|/|后窗均值|；range=(后窗最大-最小)/|后窗均值|。
     if(it>=Integer.parseInt(p.getProperty("minimumIterations")) && ds.size()>=2*window){int n=ds.size();double a=0,b=0,mn=Double.POSITIVE_INFINITY,mx=Double.NEGATIVE_INFINITY;for(int j=n-2*window;j<n-window;j++)a+=ds.get(j)/window;for(int j=n-window;j<n;j++){double z=ds.get(j);b+=z/window;mn=Math.min(mn,z);mx=Math.max(mx,z);}drift=Math.abs(a-b)/Math.max(Math.abs(b),1e-9);range=(mx-mn)/Math.max(Math.abs(b),1e-9);resmax=0;for(Monitor mon:s.getMonitorManager().getObjects())if(mon instanceof ResidualMonitor){double[] vals=((ResidualMonitor)mon).getAllYValues();if(vals.length>0)resmax=Math.max(resmax,Math.abs(vals[vals.length-1]));}
      // 四项条件须同时满足才提前停止；这是本算例的数值收敛标准，不代表物理精度已验证。
      if(drift<Double.parseDouble(p.getProperty("relativeDragDrift")) && range<Double.parseDouble(p.getProperty("relativeDragRange")) && mass<Double.parseDouble(p.getProperty("relativeMassImbalance")) && resmax<Double.parseDouble(p.getProperty("maximumNormalizedResidual"))){converged=true;break;}
     }
    }
   }
   // 更新残差、表面场和最终仿真文件，使这些产物对应同一个最终迭代状态。
   for(Monitor mon:s.getMonitorManager().getObjects())if(mon instanceof ResidualMonitor)mon.export(folder+"/residual_"+mon.getPresentationName().replaceAll("[^A-Za-z0-9_-]","_")+".csv");
   XyzInternalTable table=(XyzInternalTable)s.getTableManager().getTable("VF_SurfacePressure");table.extract();table.export(folder+"/surface.csv",",");s.saveState(folder+"/final_result.sim");
   // 将是否收敛和实际指标一起落盘；启动器/后处理读取此文件，无需人工抄数。
   try(PrintWriter out=new PrintWriter(new FileWriter(folder+"/convergence.json"))){out.printf(Locale.US,"{\"converged\":%s,\"iteration\":%d,\"drag_relative_drift\":%.10g,\"drag_relative_range\":%.10g,\"mass_imbalance_relative\":%.10g,\"max_final_normalized_residual\":%.10g}%n",converged,s.getSimulationIterator().getCurrentIteration(),drift,range,mass,resmax);}
   // 完成标记表示宏正常执行到底；还必须查看 CONVERGED 的值。
   s.println("VF_STAGE=CASE_COMPLETE CONVERGED="+converged);
  }catch(Exception e){s.println("VF_ERROR="+e);throw new RuntimeException(e);}
 }
}
